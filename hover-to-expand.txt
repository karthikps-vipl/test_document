  <script>
    // Initialize Lucide Icons
    lucide.createIcons();

    // Active Layout State
    let currentLayout = 'drawer'; // 'drawer' | 'stack' | 'tooltip'
    let currentTheme = 'charcoal'; // 'charcoal' | 'darkblue'

    // Sidebar references
    const sidebar = document.getElementById('sample-sidebar');
    const navItems = document.querySelectorAll('.nav-item');
    const navLabels = document.querySelectorAll('.nav-label');
    const profileLabel = document.getElementById('profile-lbl');
    const tooltips = document.querySelectorAll('.nav-tooltip');

    /**
     * Swaps structural layout configurations
     */
    function switchLayout(layoutType) {
      currentLayout = layoutType;
      
      // Reset active selector buttons
      const buttons = ['drawer', 'stack', 'tooltip'];
      buttons.forEach(btn => {
        const el = document.getElementById(`btn-${btn}`);
        if (btn === layoutType) {
          el.className = "group text-left p-5 rounded-2xl border bg-slate-900/50 border-indigo-500/30 ring-1 ring-indigo-500/20 transition-all duration-250 hover:bg-slate-900";
          el.querySelector('span').className = "text-xs font-semibold tracking-wider text-indigo-400 uppercase";
          // Safer class selector that avoids literal dot parsing issues
          const iconWrapper = el.querySelector('.icon-container');
          if (iconWrapper) {
            iconWrapper.className = "icon-container p-1.5 rounded-lg bg-indigo-500/10 text-indigo-400";
          }
        } else {
          el.className = "group text-left p-5 rounded-2xl border bg-slate-900/20 border-slate-800 transition-all duration-250 hover:bg-slate-900 hover:border-slate-700";
          el.querySelector('span').className = "text-xs font-semibold tracking-wider text-slate-500 uppercase";
          // Safer class selector that avoids literal dot parsing issues
          const iconWrapper = el.querySelector('.icon-container');
          if (iconWrapper) {
            iconWrapper.className = "icon-container p-1.5 rounded-lg bg-slate-800 text-slate-400";
          }
        }
      });

      // Tear down all layouts to clean slate first
      clearLayoutStyles();

      // Configure selected layout
      if (layoutType === 'drawer') {
        setupDrawerLayout();
      } else if (layoutType === 'stack') {
        setupStackLayout();
      } else if (layoutType === 'tooltip') {
        setupTooltipLayout();
      }
    }

    function clearLayoutStyles() {
      // 1. Reset Sidebar Base Width
      sidebar.className = "drawer-transition h-[92vh] rounded-[24px] border border-white/[0.06] flex flex-col justify-between py-5 text-slate-400 select-none shadow-2xl relative";
      sidebar.classList.add('w-[72px]');
      sidebar.style.removeProperty('width');

      // Remove Event Listeners by cloning elements if needed, but manual variable control is cleaner
      sidebar.onmouseenter = null;
      sidebar.onmouseleave = null;

      // 2. Clear Label structures
      navLabels.forEach(lbl => {
        lbl.className = "nav-label drawer-transition opacity-0 hidden pr-3 text-xs font-medium truncate";
      });
      profileLabel.className = "drawer-transition opacity-0 hidden pr-3 text-xs font-semibold text-white truncate max-w-[110px]";

      // 3. Clear stack styling classes
      navItems.forEach(item => {
        item.className = "nav-item group flex items-center justify-end w-full rounded-xl cursor-pointer py-2.5 transition-all duration-200 px-3 relative";
        // restore correct active state styling manually
        if (item.getAttribute('data-nav-id') === 'search') {
          item.classList.add('text-indigo-400', 'bg-indigo-500/10');
        } else if (item.getAttribute('data-nav-id') === 'toggle-ctrl') {
          item.className = "nav-item flex items-center justify-end w-full rounded-xl cursor-pointer py-2 px-3 relative";
        } else {
          item.classList.add('hover:text-white', 'hover:bg-white/[0.04]');
        }

        // Reset alignment styles
        const icon = item.querySelector('i');
        if (icon) {
          icon.className = "w-5 h-5 flex-shrink-0";
        }
      });

      // Reset bottom visual switch state to collapsed style
      const switchTrack = document.getElementById('visual-switch-track');
      switchTrack.className = "w-10 h-6 bg-[#333] rounded-full p-0.5 transition duration-200 flex items-center justify-end flex-shrink-0";

      // Hide tooltips
      tooltips.forEach(t => {
        t.className = "nav-tooltip absolute right-[76px] bg-slate-900 text-slate-100 text-[11px] font-medium px-2.5 py-1.5 rounded-lg border border-slate-800 shadow-xl opacity-0 translate-x-2 pointer-events-none transition-all duration-200 whitespace-nowrap hidden";
      });
    }

    /**
     * OPTION 1: HOVER-TO-EXPAND DRAWER
     */
    function setupDrawerLayout() {
      // Dynamic expansion on hover
      sidebar.onmouseenter = () => {
        sidebar.style.width = '190px';
        
        // Show labels (wait for width expansion)
        setTimeout(() => {
          if (currentLayout !== 'drawer') return;
          
          navLabels.forEach(lbl => {
            lbl.classList.remove('hidden');
            setTimeout(() => lbl.classList.remove('opacity-0'), 20);
          });
          
          profileLabel.classList.remove('hidden');
          setTimeout(() => profileLabel.classList.remove('opacity-0'), 20);

          // Update active layout of toggles
          const switchLbl = document.querySelector('[data-nav-id="toggle-ctrl"] .nav-label');
          if (switchLbl) {
            switchLbl.classList.remove('hidden');
            switchLbl.classList.remove('opacity-0');
          }
        }, 120);
      };

      sidebar.onmouseleave = () => {
        sidebar.style.width = '72px';
        
        // Instant hide to prevent wrapping
        navLabels.forEach(lbl => {
          lbl.classList.add('opacity-0', 'hidden');
        });
        profileLabel.classList.add('opacity-0', 'hidden');

        const switchLbl = document.querySelector('[data-nav-id="toggle-ctrl"] .nav-label');
        if (switchLbl) {
          switchLbl.classList.add('opacity-0', 'hidden');
        }
      };
    }

    /**
     * OPTION 2: VERTICAL STACK
     */
    function setupStackLayout() {
      // Set fixed slightly wider profile to fit micro-text comfortably
      sidebar.style.width = '74px';

      // Re-configure layout into vertical stacks
      navItems.forEach(item => {
        const id = item.getAttribute('data-nav-id');
        
        // Ignore the dark-mode switch row structure
        if (id === 'toggle-ctrl') {
          item.className = "nav-item flex flex-col items-center justify-center w-full rounded-xl cursor-pointer py-1 px-1 relative";
          const switchTrack = document.getElementById('visual-switch-track');
          switchTrack.className = "w-8 h-4 bg-[#333] rounded-full p-0.5 transition duration-200 flex items-center justify-end flex-shrink-0";
          switchTrack.querySelector('div').className = "w-3 h-3 bg-white rounded-full shadow-md transition duration-200";
          return;
        }

        item.className = "nav-item group flex flex-col items-center justify-center w-full rounded-xl cursor-pointer py-2 transition-all duration-200 px-1 relative text-center";
        
        // Color assignment
        if (id === 'search') {
          item.classList.add('text-indigo-400', 'bg-indigo-500/5');
        } else if (id === 'logout') {
          item.classList.add('hover:text-rose-400', 'hover:bg-rose-500/10');
        } else {
          item.classList.add('hover:text-white', 'hover:bg-white/[0.04]');
        }

        // Micro-Text modification
        const labelText = item.querySelector('.nav-label');
        if (labelText) {
          labelText.className = "nav-label text-[9px] font-bold tracking-wider uppercase mt-1.5 opacity-100 flex-shrink-0 text-slate-400 group-hover:text-white block truncate max-w-[62px]";
          // Custom override clean names for stack format
          if (id === 'search') labelText.innerText = 'Search';
          if (id === 'home') labelText.innerText = 'Home';
          if (id === 'stats') labelText.innerText = 'Stats';
          if (id === 'alerts') labelText.innerText = 'Alerts';
          if (id === 'reports') labelText.innerText = 'Reports';
          if (id === 'favs') labelText.innerText = 'Favs';
          if (id === 'wallet') labelText.innerText = 'Wallet';
          if (id === 'logout') labelText.innerText = 'Exit';
        }

        // Shrink the icons slightly for balanced stack proportion
        const icon = item.querySelector('i');
        if (icon) {
          icon.className = "w-4 h-4 flex-shrink-0";
        }
      });
    }

    /**
     * OPTION 3: TOOLTIP POPOUT (DYNAMIC LEFT SHIFT)
     */
    function setupTooltipLayout() {
      // Keeps menu slim
      sidebar.style.width = '64px';

      // Attach custom tooltip triggers
      navItems.forEach(item => {
        const id = item.getAttribute('data-nav-id');
        if (id === 'toggle-ctrl') return;

        const tooltip = item.querySelector('.nav-tooltip');
        if (!tooltip) return;

        tooltip.classList.remove('hidden');

        item.onmouseenter = () => {
          if (currentLayout !== 'tooltip') return;
          tooltip.classList.remove('opacity-0', 'translate-x-2');
          tooltip.classList.add('opacity-100', 'translate-x-0');
        };

        item.onmouseleave = () => {
          if (currentLayout !== 'tooltip') return;
          tooltip.classList.remove('opacity-100', 'translate-x-0');
          tooltip.classList.add('opacity-0', 'translate-x-2');
        };
      });
    }

    /**
     * Toggles layout theme colors
     */
    function toggleSidebarTheme() {
      const label = document.getElementById('theme-status');
      if (currentTheme === 'charcoal') {
        currentTheme = 'darkblue';
        sidebar.style.backgroundColor = '#0b1329';
        sidebar.style.borderColor = 'rgba(59, 130, 246, 0.15)';
        label.innerText = 'Midnight Blue';
      } else {
        currentTheme = 'charcoal';
        sidebar.style.backgroundColor = '#1e1e1e';
        sidebar.style.borderColor = 'rgba(255, 255, 255, 0.06)';
        label.innerText = 'Charcoal Dark';
      }
    }

    // Default initialization
    setupDrawerLayout();
